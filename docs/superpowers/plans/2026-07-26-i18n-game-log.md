# i18n-Translatable Game Log Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Change `GET /api/v1/game/{roomCode}/log` from returning pre-baked English sentences to
structured `{ type, data }` entries, so the frontend's "View full game log" screen can render them
in the active locale (English or Vietnamese) instead of always showing English prose.

**Architecture:** Backend (`../werewolf`): `GameLogView`'s entries become a typed record instead of
a formatted string, built the same way (one `Apply` overload per domain event), and the endpoint
stops doing server-side player-name string substitution. Frontend (this repo): a new pure-function
util maps each entry's `type` to a translation key, resolving player ids to names and enum-ish
fields (role/faction/cause) to their existing `roles.*`/`factions.*`/new `gameLog.causes.*` keys —
mirroring exactly how the faction-name bug was fixed earlier this session.

**Tech Stack:** Backend: ASP.NET Core + WolverineFx.Http + Marten (C#, .NET 10). Frontend: Angular
22 signals + `@ngx-translate/core`.

## Global Constraints

- No new HTTP routes, no new SignalR commands — only the shape of the existing `GET
/api/v1/game/{roomCode}/log` response changes.
- Every one of the 21 `GameLogViewProjection.Apply`/`Create` overloads in
  `../werewolf/src/Application/Werewolf/ReadModels/GameLogView.cs` must be covered — no event type
  silently dropped from the log.
- Per `CLAUDE.md`: any change to an HTTP response shape must update `GAME_FLOW.md` (this repo),
  `docs/api-and-signalr-commands.md`, and `../werewolf/GAME_FLOW.md` in the same
  commit/PR — not a follow-up.
- Never add a `Co-Authored-By` trailer to any commit in either repo.
- Backend build/run: `../werewolf`'s own `.claude/skills/run-werewolf/driver.sh up` (Linux/CI) — on
  this Windows box, build directly with `dotnet build src/Application/Application.csproj` from
  `../werewolf`, and if `dotnet run` is needed, follow this session's earlier working sequence:
  ensure `podman compose up -d` (from `../werewolf`) has Postgres up, then `dotnet run
--no-launch-profile --urls http://localhost:5080` from `../werewolf/src/Application` with
  `ASPNETCORE_ENVIRONMENT=Development` set. If the build fails with a stale-codegen error pointing

```csharp
using Application.Werewolf.ReadModels;
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace Application.Werewolf.Game.GetGameLog;

public record GameLogResponse
{
    public required string RoomCode { get; init; }
    public required Guid GameId { get; init; }
    public required List<GameLogEntry> Entries { get; init; }
}

// Backs the frontend's Game Over "View full game log" screen. Entries are structured
// (type + raw data) rather than pre-formatted English sentences specifically so the frontend can
// render them in the active locale -- player names, role/faction names, and death causes are all
// resolved client-side from ids/enum values, never baked into a string here.
public static class GetGameLogEndpoint
{
    [WolverineGet("/api/v1/game/{roomCode}/log")]
    public static async Task<GameLogResponse?> Handle(
        RoomCode roomCode,
        IDocumentSession session,
        CancellationToken cancellationToken)
    {
        var state = await session.Events.FetchLatest<GameState, RoomCode>(roomCode, cancellationToken);
        if (state is null)
        {
            return null;
        }

        var log = await session.LoadAsync<GameLogView>(state.Id, cancellationToken);
        if (log is null)
        {
            return null;
        }

        return new GameLogResponse
        {
            RoomCode = state.RoomCode.Value,
            GameId = state.Id,
            Entries = log.Entries
        };
    }
}
```

This drops the `PlayerDirectoryProjection`/`namesById`/`.Replace()` block entirely (no longer
needed — the frontend resolves names) along with the `using Application.Werewolf.Domain;` and
`using System.Linq;` usings it required, which are otherwise unused in this file now.

```ts
import { TranslateService } from '@ngx-translate/core';
import { GameLogEntry } from '../models/game.model';

export interface GameLogRenderContext {
    translate: TranslateService;
    playerName: (playerId: string) => string;
}

function name(ctx: GameLogRenderContext, data: GameLogEntry['data'], key: string): string {
    const id = data[key];
    return typeof id === 'string' ? ctx.playerName(id) : '';
}

/** Renders one structured game log entry into the active locale's display string -- player ids
 * resolve through `ctx.playerName` (never a raw id shown to the player), and role/faction/cause
 * enum-ish values resolve through their existing `roles.*`/`factions.*`/`gameLog.causes.*`
 * translation keys instead of being interpolated raw (the same class of bug the Game Over
 * banner's winning-faction text had before it was fixed earlier this session). */
export function renderGameLogEntry(entry: GameLogEntry, ctx: GameLogRenderContext): string {
    const { translate } = ctx;
    const { type, data } = entry;

            });
        case 'seer_inspection_performed':
            return translate.instant('gameLog.seerInspectionPerformed', {
                seer: name(ctx, data, 'seerPlayerId'),
                target: name(ctx, data, 'targetPlayerId'),

                )
            });
        case 'doctor_protection_chosen':

                doctor: name(ctx, data, 'doctorPlayerId'),

            });
        case 'werewolf_target_locked':

            });
        case 'werewolf_no_kill':

        case 'witch_heal_used':

                witch: name(ctx, data, 'witchPlayerId')
            });
        case 'witch_poison_used':
            return translate.instant('gameLog.witchPoisonUsed', {
                witch: name(ctx, data, 'witchPlayerId'),

        case 'witch_passed':

        case 'voting_started':
            return translate.instant('gameLog.votingStarted');
        case 'vote_cast':

            });
        case 'player_lynched':
            return translate.instant('gameLog.playerLynched', {
                name: name(ctx, data, 'playerId')
            });
        case 'no_lynch_occurred':
            return translate.instant('gameLog.noLynchOccurred');
        case 'hunter_revenge_pending':
            return translate.instant('gameLog.hunterRevengePending', {
                hunter: name(ctx, data, 'hunterPlayerId')
            });
        case 'hunter_revenge_shot_fired':
            return translate.instant('gameLog.hunterRevengeShotFired', {

            });
        case 'hunter_revenge_declined':
            return translate.instant('gameLog.hunterRevengeDeclined', {

            });
        case 'game_ended':
            return translate.instant('gameLog.gameEnded', {

        default:
            return '';
    }
}
```
